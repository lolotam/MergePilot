## Proposed contracts

Create `packages/webscan/src/types.ts`. Names and units below are the contract shared by the package implementation tasks; API/store types may add ids and timestamps without putting database concerns into the package.

```ts
export type CheckId = 'headers' | 'cookies' | 'transport' | 'disclosure' | 'files' | 'active';
export type Validation = 'Confirmed' | 'Likely' | 'Indicator' | 'Informational';
export type Comparison = 'New' | 'Unchanged' | 'Fixed' | 'Not checked';
export type Severity = 'critical' | 'high' | 'medium' | 'low' | 'info';
export type CheckSwitches = Record<CheckId, boolean>;

export interface ScanLimits {
  maxRequests: number;
  maxPages: number;
  maxActiveLinks: number;
  requestsPerSecond: number;
  requestTimeoutMs: number;
  maxResponseBytes: number;
  totalTimeoutMs: number;
}

export interface PathRule { path: string; kind: 'exact' | 'prefix' }
export interface ParameterRule { path: string; parameters: readonly string[] }
export interface ScanScope {
  hostname: string;
  baseUrl: string;
  crawlPaths: readonly PathRule[];
  blockedPaths: readonly PathRule[];
  parameterizedPaths: readonly ParameterRule[];
  activePaths: readonly ParameterRule[];
  extraFilePaths: readonly string[];
}

export interface ScanInput {
  scope: ScanScope;
  limits: ScanLimits;
  checks: CheckSwitches;
  contact: string;
}

export interface ScanProgress {
  requests: number;
  pages: number;
  elapsedMs: number;
  phase: 'crawl' | 'checks' | 'complete';
}

export interface ScanDependencies {
  // Requests contain transient raw URLs. Implementations must never log them.
  // HEAD is allowed by the transport rules, but exposed-file checks always use
  // the size-capped GET (ADR-0052 section 3).
  fetch: (url: string, init: { method: 'GET' | 'HEAD'; signal: AbortSignal }) => Promise<Response>;
  now: () => number;
  sleep: (milliseconds: number, signal: AbortSignal) => Promise<void>;
  signal: AbortSignal;
  onProgress: (progress: ScanProgress) => void;
  // App-supplied per-site HMAC functions; raw inputs never leave this process.
  identifyTarget: (baselineUrl: string) => string;
  identifyOccurrence: (input: {
    targetKey: string; checkId: CheckId; issueId: string; subject: string;
  }) => string;
}
```

The injected `fetch` must apply the site's validated hostname, pinned address, size/time caps and live worker authorization/lock checks. `runScan` independently enforces the same host/scheme/port and request-budget restrictions. Do not expose an option that bypasses public-address validation in production.

The package returns only safe catalogue findings, occurrence identities, normalized structural evidence and coverage. Raw `Response`/HTML/URLs with values never escape into the result. The app performs its additional engine redaction before persisting the result. Reports consume the stored safe result, not raw transport data.

Scanning and historical comparison are separate operations. Define these proposed exports in `types.ts`, `run.ts` and `compare.ts`:

```ts
export interface OccurrenceFinding {
  occurrenceKey: string;
  targetKey: string;
  checkId: CheckId;
  checkVersion: number;
  issueKey: string;
  issueId: string;
  subject: string;
  severity: Severity;
  validation: Validation;
  confidence: 'high' | 'medium' | 'low';
  safeLocation: string;
  evidenceHash: string;
  // Only catalogue-approved, structurally sanitized facts.
  // `null` is an explicit state, not an empty finding: it means retention pruned
  // the evidence payloads and only the identity plus the exception-pinned
  // evidenceHash remain (an outstanding row restored from an archived tombstone).
  evidence: Readonly<Record<string, string | number | boolean>> | null;
}
export interface CoverageEntry {
  targetKey: string;
  checkId: CheckId;
  checkVersion: number;
  evaluatedIssueIds: readonly string[];
  outcome: 'checked' | 'error' | 'not-visited';
  reason?: 'redirect_out_of_scope' | 'http_probe_unreachable' | 'disabled' | 'not_visited' | 'target_error';
}
export type ScanStopReason = 'request_limit' | 'time_limit' | 'rate_limited'
  | 'cancelled' | 'kill_switch' | 'authorization_revoked' | 'lease_lost';
export type ScanFailureReason = 'target_error' | 'transport_error' | 'policy_violation' | 'worker_lost';
export interface ScanObservationData {
  partialCoverage: boolean;
  requests: number;
  pages: number;
  durationMs: number;
  findings: readonly OccurrenceFinding[];
  coverage: readonly CoverageEntry[];
}
export type ScanObservation = ScanObservationData & (
  | { status: 'completed'; reason?: never }
  | { status: 'stopped'; reason: ScanStopReason }
  | { status: 'failed'; reason: ScanFailureReason }
);
export interface RetainedOccurrenceIdentity {
  occurrenceKey: string;
  targetKey: string;
  checkId: CheckId;
  checkVersion: number;
  issueKey: string;
  issueId: string;
  subject: string;
  // The retained identity is every safe OccurrenceFinding field except
  // `evidence`: severity, validation, confidence and safeLocation survive
  // archival, so a tombstone can be restored or proven Fixed without
  // fabricating a value. Only the evidence body is pruned.
  severity: Severity;
  validation: Validation;
  confidence: 'high' | 'medium' | 'low';
  safeLocation: string;
  evidenceHash: string;
}
export type RetainedOccurrence =
  | { finding: OccurrenceFinding; state: 'outstanding' }
  | { finding: OccurrenceFinding; state: 'fixed' }
  | { identity: RetainedOccurrenceIdentity; state: 'archived';
      archiveReason: 'unobserved_retention' | 'retired_catalogue' };
export interface ComparisonResult {
  mayBecomeBaseline: boolean;
  updates: readonly (
    | { finding: OccurrenceFinding; comparison: Comparison; state: 'outstanding' }
    | { finding: OccurrenceFinding; comparison: Comparison; state: 'fixed' }
    | { identity: RetainedOccurrenceIdentity; comparison: 'Not checked'; state: 'archived';
        archiveReason: 'unobserved_retention' | 'retired_catalogue' }
  )[];
}
export declare function runScan(input: ScanInput, deps: ScanDependencies): Promise<ScanObservation>;
export declare function compareScan(
  current: ScanObservation,
  retained: readonly RetainedOccurrence[],
): ComparisonResult;
```

The app loads retained occurrence state from storage — outstanding rows plus archival tombstones — and supplies both to `compareScan` after the scan. An archived tombstone is metadata only (`RetainedOccurrenceIdentity`, no `finding` and no evidence body) and remains historical Not checked unless the current terminal result re-observes its exact occurrence key: a new finding re-emits such a row as `outstanding` — direct positive evidence, permitted from any terminal run (a stopped run is not a baseline and still re-activates) — and a successful negative from a **completed** run emits it as `fixed`, which is how a group blocked by its archived-unverified members finally reaches its deferred Fixed comment alongside the ADR-0052 §11 administrator-resolution end state. Both transitions are built from the retained identity, never from a fabricated finding: `RetainedOccurrenceIdentity` carries every safe `OccurrenceFinding` field except `evidence`, so the emitted `OccurrenceFinding` takes its severity, validation, confidence and safe location from the retained metadata and sets `evidence: null` to represent the pruned body explicitly. The same construction restores a tombstone outside the scan path: when an administrator resolution over it expires or is revoked, the store rebuilds the outstanding `OccurrenceFinding` from the retained identity with `evidence: null` (ADR-0052 §11). There is no automatic group-level path for churned variants: raw query values are never persisted, so a scan that never re-observes that exact key can neither evaluate nor prove that member, and only the administrator unblocks it. Coverage `checked` means the listed issue ids were fully evaluated for that exact target, including absence of a previously reported subject; fetching only headers does not cover body checks, and a scope-denied redirect supplies checked coverage only for the ids evaluated on the 3xx response itself (ADR-0052 §2). Comparison can infer Fixed only for an outstanding or archived occurrence absent from current findings with matching target/check and successful coverage whose `evaluatedIssueIds` includes that immutable issue id, in a completed run. A newer check implementation may provide this coverage without changing occurrence identity; meaning changes require a new issue id, and a retired id is emitted as `archived` with `archiveReason: 'retired_catalogue'` instead of ever being converted to Fixed by a version change. It preserves old outstanding records as Not checked otherwise. Keys use the original rediscovered baseline URL, so another query variant cannot remediate it. No retained database connection or raw query values cross this API boundary.
